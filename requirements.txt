# Mongla -- third-party Python packages (pip)
# ----------------------------------------------------------------------
# What the code imports that ROS does not provide. ROS packages (rclpy,
# cv_bridge, sensor_msgs, ...) come from apt: ROS 2 Jazzy on the vehicle
# (Raspberry Pi 5, Ubuntu 24.04), Humble on dev boxes.
#
#     pip install -r requirements.txt            # dev box
#
# The vehicle does NOT use this file: kutum/setup.sh installs it from
# kutum/requirements.txt, with the exact pins that hold on Ubuntu 24.04. Three rules there, each a
# silent failure if broken (.claude/context/kutum/pi-and-env-traps.md):
#   * OpenCV is the SYSTEM cv2 (apt python3-opencv). A pip opencv shadows it,
#     has no window backend, and drags numpy 2 in.
#   * numpy stays 1.x: cv_bridge and the system cv2 are built against the 1.x
#     ABI and fail with `_ARRAY_API not found` under 2.x.
#   * hailo_platform is not on PyPI: kutum/setup.sh builds it from source
#     against the installed libhailort, at the same version.
# ----------------------------------------------------------------------

# ===================== the vehicle (and every dev box) =================

# The board link: MAVLink 2 over the SROT board's USB serial.
# fc/srot_fc.py and every tools/srot_*.py.
pymavlink>=2.4.41
# serial port probing in the manager and the bench tools
pyserial>=3.5

# Pinned below 2.0 -- see the numpy rule above.
numpy>=1.24,<2.0

# Tracking (tracking/roboflow_tracker.py, bytetrack.py, kalman.py). All three
# are imported lazily, and each has a logged fallback.
# trackers: 2.4.0 exactly. The 2.5.0 wheel on PyPI contains no module, and its
# numpy>=2 requirement is spurious, so on the Pi install it with --no-deps:
#     pip install --no-deps "trackers==2.4.0"
trackers>=2.4.0,<2.5.0
# supervision: <=0.26.1 on numpy 1.x (0.30 uses np.long, which is numpy-2 only)
supervision>=0.18.0,<=0.26.1
filterpy>=2.0.0

# HUD text rendering (_text_engine.py)
Pillow>=13.0.0

# OPTIONAL. ONNX runtime for the XFeat matcher (anchor/xfeat_onnx.py) and the
# monocular depth node (depth/depth_estimation_node.py). Imported lazily; the
# Hailo path (anchor/xfeat_hailo.py) does not need it.
onnxruntime>=1.31.1

# ===================== dev box only ====================================

# .pt models: replaying archive footage on a desktop GPU (detection/yolo.py).
# The vehicle runs .hef on the Hailo-8 and never imports these.
ultralytics>=8.4.203; platform_machine == "x86_64"
# pip OpenCV is fine on x86 dev boxes; on the Pi see the OpenCV rule above.
opencv-python>=4.8; platform_machine == "x86_64"

# plots in tools/ (scorers, sweeps)
matplotlib>=3.11.2; platform_machine == "x86_64"

# tests
pytest>=9.2.0
